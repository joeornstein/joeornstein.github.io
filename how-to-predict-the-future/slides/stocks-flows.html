P0 = transpose(pop)
tfr0 = d3.sum(P0, d => d.fert)            // today's total fertility rate (births per woman)

// births from today's women, then deaths, then everyone who survives gets a year older
step = (st, par) => {
  const n = st.m.length, m = new Array(n), f = new Array(n);
  const births = d3.sum(st.f, (x, i) => x * par.fert[i]);
  m[0] = births / 2; f[0] = births / 2;
  for (let i = 1; i < n; i++) {
    m[i] = st.m[i - 1] * (1 - par.qm[i - 1]);
    f[i] = st.f[i - 1] * (1 - par.qf[i - 1]);
  }
  m[n - 1] += st.m[n - 1] * (1 - par.qm[n - 1]);   // the 100+ group is open-ended
  f[n - 1] += st.f[n - 1] * (1 - par.qf[n - 1]);
  return {m, f};
}

makePar = (tfr = tfr0, dmult = 1) => ({
  fert: P0.map(d => d.fert * tfr / tfr0),
  qm: P0.map(d => Math.min(1, d.qm * dmult)),
  qf: P0.map(d => Math.min(1, d.qf * dmult))
})

today = ({m: P0.map(d => d.m), f: P0.map(d => d.f)})

// summary statistics for a population
describe = (st) => {
  const tot = st.m.map((x, i) => x + st.f[i]);
  const N = d3.sum(tot), share = tot.map(x => x / N);
  let cum = 0, median = 0;
  for (let i = 0; i < share.length; i++) {
    if (cum + share[i] >= 0.5) { median = i + (0.5 - cum) / share[i]; break; }
    cum += share[i];
  }
  const sumAges = (lo, hi) => d3.sum(tot.slice(lo, hi + 1));
  return {N, share, median, old: sumAges(65, 100) / N, workers: sumAges(18, 64) / sumAges(65, 100)};
}

// life expectancy at birth implied by the death rates (average of men & women)
lifeExp = (par) => {
  const e0 = (q) => {
    let l = 1, e = 0;
    for (let i = 0; i < q.length - 1; i++) { const next = l * (1 - q[i]); e += (l + next) / 2; l = next; }
    return e + l / q[q.length - 1];
  };
  return (e0(par.qm) + e0(par.qf)) / 2;
}

ageData = (share) => share.map((s, i) => ({age: i, share: s}))
todayStats = describe(today)
todayE0 = lifeExp(makePar())

Stocks & Flows

  • A stock measures some quantity at a point in time.

  • A flow measures the rate of change in a quantity.

  • A useful metaphor is a bathtub (Sterman 2000).

Stocks & Flows

Lots of problems can be described in terms of stocks and flows:

Stock Inflow Outflow
$ in my bank account Monthly income Monthly spending
National Debt Taxes Government Spending
Number of Democracies Democratization “Democratic Backsliding”
CO2 in Atmosphere Carbon Emissions Carbon Sinks
Population Births Deaths

  • Confusing stocks and flows is a common mistake.

  • For example: debt is a stock, while deficit is a flow.

    • “Cutting the deficit” does not necessarily mean that the national debt will decrease!

Markov Models

  • When flows are defined in terms of probabilities, we call it a Markov Model.
  • For example, suppose that every autocracy has a 4% chance of democratizing each year.
  • And every democracy has a 1% chance of becoming an autocracy…
  • If this process continued for 1,000 years, what do you think would happen to the share of countries that are democracies?

Equilibrium

  • Eventually, the system settles into a stable point, called an equilibrium.

  • In equilibrium, inflows = outflows. Nothing changes from year to year.

  • Intuitively, if the rate of democratization (4% per year) is 4x the rate of backsliding (1% per year), then in equilibrium there were be 4x more democracies than autocracies.

  • 4:1 = 80%

Does the Starting Point Matter?

  • What if we started with 100% democracies? Or a 50/50 split?
  • Doesn’t matter! Regardless of starting point, the system converges to the same equilibrium: 80% democracies.

  2. A recession hits, and the job-finding rate falls to 18% per month. What is the new equilibrium unemployment rate?

Ergodic Theorem

  • When should we expect a system of stocks & flows to reach an equilibrium?

  • The ergodic theorem states that there will be a unique equilibrium for any system that meets the following conditions:

    • There exists a path from every stock to every other stock.

    • Flow probabilities are stationary (i.e. unchanging).

  • No matter how complicated the model gets, if these two conditions hold, there will be a unique equilibrium where all the inflows equal outflows.

// the equilibrium: run the model forward for a long, long time
eqm = {
  const par = makePar(tfr, dmult);
  let st = today;
  for (let t = 1; t <= 1000; t++) {
    st = step(st, par);
    const N = d3.sum(st.m) + d3.sum(st.f);          // rescale each year: only the shares matter
    st = {m: st.m.map(x => x / N), f: st.f.map(x => x / N)};
  }
  return {stats: describe(st), e0: lifeExp(par)};
}

Takeaways

  • Keep in mind the distinction between stocks (the level) and flows (the rate of change).

  • A system of stocks and flows is ergodic if the flow probabilities remain fixed and there’s a path from every stock to every other stock.

    • Ergodic systems always converge to an equilibrium state.

  • Next time, we’ll discuss positive and negative feedback.

    • i.e. What happens when the size of a stock affects the rate of inflow/outflow?
